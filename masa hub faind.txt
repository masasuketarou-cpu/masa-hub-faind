local function isBlacklisted(obj)
	for _, name in ipairs(blacklist) do
		if obj:IsA(name) then
			return true
		end
	end
	return false
end

local function clearEffects()
	for _, v in pairs(Lighting:GetDescendants()) do
		if isBlacklisted(v) then
			v:Destroy()
		end
	end
end

-- Anti Bee を有効にする関数
local function enableAntiBee()
	if antiBeeActive then return end
	antiBeeActive = true
	
	-- 既存のエフェクトをクリア
	clearEffects()
	
	-- DescendantAdded 接続
	antiBeeDescendantConnection = Lighting.DescendantAdded:Connect(function(obj)
		task.wait()
		if isBlacklisted(obj) then
			obj:Destroy()
		end
	end)
	table.insert(antiBeeConnections, antiBeeDescendantConnection)
	
	-- FOV ロック接続
	local fovConnection = RunService.RenderStepped:Connect(function()
		if camera.FieldOfView ~= FOV_LOCK then
			camera.FieldOfView = FOV_LOCK
		end
	end)
	table.insert(antiBeeConnections, fovConnection)
	
	-- 移動入力接続
	local inputConnection = UserInputService.InputChanged:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.Gamepad1 or input.UserInputType == Enum.UserInputType.Touch then
			if input.KeyCode == Enum.KeyCode.Thumbstick1 then
				moveVector = Vector3.new(input.Position.X, 0, -input.Position.Y)
			end
		end
	end)
	table.insert(antiBeeConnections, inputConnection)
	
	-- 移動適用接続
	local moveConnection = RunService.RenderStepped:Connect(function()
		local char = plr.Character
		if char and char:FindFirstChildOfClass("Humanoid") then
			local humanoid = char:FindFirstChildOfClass("Humanoid")
			humanoid:Move(moveVector, true)
		end
	end)
	table.insert(antiBeeConnections, moveConnection)
end

-- Anti Bee を無効にする関数
local function disableAntiBee()
	if not antiBeeActive then return end
	antiBeeActive = false
	
	-- 全ての接続を切断
	for _, connection in ipairs(antiBeeConnections) do
		if connection then
			connection:Disconnect()
		end
	end
	antiBeeConnections = {}
	antiBeeDescendantConnection = nil
	
	-- moveVector リセット
	moveVector = Vector3.zero
end

-- Anti Bee 切り替え関数
function flipAntiBee(state)
    if state then
        antiBeeActive = true
        antiBeeBtn.BackgroundColor3 = Color3.fromRGB(120, 0, 0)
        antiBeeBorder.Color = Color3.fromRGB(255, 0, 0)
        antiBeeText.Text = "Anti Bee: ON"
        antiBeeText.TextColor3 = Color3.fromRGB(255, 60, 60)
        enableAntiBee()
    else
        antiBeeActive = false
        antiBeeBtn.BackgroundColor3 = Color3.fromRGB(30, 0, 0)
        antiBeeBorder.Color = Color3.fromRGB(180, 0, 0)
        antiBeeText.Text = "Anti Bee: OFF"
        antiBeeText.TextColor3 = Color3.fromRGB(255, 100, 100)
        disableAntiBee()
    end
end

-- ボタンクリックイベント
antiBeeBtn.MouseButton1Click:Connect(function()
    flipAntiBee(not antiBeeActive)
end)

-- CtrlキーでGUI表示切り替え

    end

        frameThing.Visible = not frameThing.Visible
    end
end)

-- 終了時の処理
